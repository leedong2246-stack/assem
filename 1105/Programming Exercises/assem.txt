
1. FindLargest Procedure
Create a procedure named FindLargest that receives two parameters: a pointer to a signeddoubleword array, and a count of the array’s length. The procedure must return the value of the largest array member in EAX. Use the PROC directive with a parameter list when declaring the procedure. Preserve all registers (except EAX) that are modified by the procedure. Write a test program that calls FindLargest and passes three different arrays of different lengths. Be sure to include negative values in your arrays. Create a PROTO declaration for FindLargest.
-> 
INCLUDE Irvine32.inc

FindLargest PROTO, pArray:PTR SDWORD, count:DWORD

.data
array1 SDWORD -10, -5, -20, -1, -15
array2 SDWORD 100, -200, 500, 300, -50, 400
array3 SDWORD -50, -30, -10, 0, -5

.code
main PROC
    ; 테스트 1
    INVOKE FindLargest, ADDR array1, LENGTHOF array1
    call WriteInt
    call Crlf

    ; 테스트 2
    INVOKE FindLargest, ADDR array2, LENGTHOF array2
    call WriteInt
    call Crlf

    ; 테스트 3
    INVOKE FindLargest, ADDR array3, LENGTHOF array3
    call WriteInt
    call Crlf

    exit
main ENDP

;---------------------------------------------------
; FindLargest
;---------------------------------------------------
FindLargest PROC USES ecx esi, pArray:PTR SDWORD, count:DWORD
    mov esi, pArray
    mov ecx, count
    mov eax, [esi]          ; 첫 번째 요소를 최댓값 초기값으로 설정
    add esi, 4
    dec ecx
    jz Done

L1:
    cmp [esi], eax          ; 부호 있는 비교
    jle Skip
    mov eax, [esi]          ; 더 큰 값 발견 시 갱신

Skip:
    add esi, 4
    loop L1

Done:
    ret
FindLargest ENDP
END main


2. Chess Board
Write a program that draws an 8 x 8 chess board, with alternating gray and white squares. You can use the SetTextColor and Gotoxy procedures from the Irvine32 library. Avoid the use of global variables, and use declared parameters in all procedures. Use short procedures that are focused on a single task. 
-> 
INCLUDE Irvine32.inc

DrawRow PROTO, row:DWORD, color1:DWORD, color2:DWORD
DrawSquare PROTO, x:DWORD, y:DWORD, color:DWORD

.code
main PROC
    mov ecx, 8              ; 8개 행
    mov edx, 0              ; 행 인덱스

BoardLoop:
    push ecx
    mov eax, edx
    and eax, 1              ; 짝수/홀수 행 판단
    jnz OddRow

    ; 짝수 행: 회색/흰색 순서
    INVOKE DrawRow, edx, gray + (black * 16), white + (white * 16)
    jmp NextRow

OddRow:
    ; 홀수 행: 흰색/회색 순서
    INVOKE DrawRow, edx, white + (white * 16), gray + (black * 16)

NextRow:
    inc edx
    pop ecx
    loop BoardLoop

    exit
main ENDP

DrawSquare PROC USES eax, x:DWORD, y:DWORD, color:DWORD
    INVOKE Gotoxy, dl, dh
    mov eax, color
    call SetTextColor
    mov al, ' '
    call WriteChar
    call WriteChar          ; 가로 비율 맞추기 위해 2칸 출력
    ret
DrawSquare ENDP

DrawRow PROC USES ecx edx, row:DWORD, color1:DWORD, color2:DWORD
    mov ecx, 4
    mov dh, BYTE PTR row
    mov dl, 0

RowLoop:
    INVOKE DrawSquare, edx, edx, color1
    add dl, 2
    INVOKE DrawSquare, edx, edx, color2
    add dl, 2
    loop RowLoop

    ret
DrawRow ENDP
END main

3. Chess Board with Alternating Colors
This exercise extends Exercise 2. Every 500 milliseconds, change the color of the colored
squares and redisplay the board. Continue until you have shown the board 16 times, using all possible 4-bit background colors. (The white squares remain white throughout.)

INCLUDE Irvine32.inc

DrawColoredBoard PROTO, colorIndex:DWORD

.code
main PROC
    mov ecx, 16             ; 16가지 배경색
    mov ebx, 0

ColorLoop:
    push ecx
    INVOKE DrawColoredBoard, ebx
    mov eax, 500            ; 500ms 지연
    call Delay
    inc ebx
    pop ecx
    loop ColorLoop

    exit
main ENDP

DrawColoredBoard PROC USES ecx edx, colorIndex:DWORD
    mov ecx, 8
    mov edx, 0

BoardLoop:
    push ecx
    mov eax, edx
    and eax, 1
    jnz OddRow

    mov eax, colorIndex
    shl eax, 4              ; 배경색 속성으로 전환
    add eax, white          ; 글자색
    call SetTextColor
    
    ; 짝수/홀수 행 출력 생략 구현
    inc edx
    pop ecx
    loop BoardLoop

    ret
DrawColoredBoard ENDP
END main

4. FindThrees Procedure
Create a procedure named FindThrees that returns 1 if an array has three consecutive values of 3 somewhere in the array. Otherwise, return 0. The procedure’s input parameter list contains a pointer to the array and the array’s size. Use the PROC directive with a parameter list when declaring the procedure. Preserve all registers (except EAX) that are modified by the procedure. Write a test program that calls FindThrees several times with different arrays.

INCLUDE Irvine32.inc

FindThrees PROTO, pArray:PTR DWORD, count:DWORD

.data
a1 DWORD 1, 2, 3, 3, 3, 4, 5
a2 DWORD 3, 3, 1, 3, 3, 2
a3 DWORD 1, 3, 3, 3

.code
main PROC
    INVOKE FindThrees, ADDR a1, LENGTHOF a1
    call WriteDec           ; 1 출력
    call Crlf

    INVOKE FindThrees, ADDR a2, LENGTHOF a2
    call WriteDec           ; 0 출력
    call Crlf

    INVOKE FindThrees, ADDR a3, LENGTHOF a3
    call WriteDec           ; 1 출력
    call Crlf

    exit
main ENDP

FindThrees PROC USES ecx esi edx, pArray:PTR DWORD, count:DWORD
    mov esi, pArray
    mov ecx, count
    cmp ecx, 3
    jb NotFound             ; 길이 3 미만이면 0 반환

    sub ecx, 2              ; 검사 횟수 = count - 2

L1:
    mov eax, [esi]
    cmp eax, 3
    jne Next

    mov eax, [esi+4]
    cmp eax, 3
    jne Next

    mov eax, [esi+8]
    cmp eax, 3
    jne Next

    mov eax, 1              ; 333 발견
    jmp Done

Next:
    add esi, 4
    loop L1

NotFound:
    mov eax, 0

Done:
    ret
FindThrees ENDP
END main


5. DifferentInputs Procedure
Write a procedure named DifferentInputs that returns EAX = 1 if the values of its three input parameters are all different; otherwise, return with EAX = 0. Use the PROC directive with a parameter list when declaring the procedure. Create a PROTO declaration for your procedure, and call it five times from a test program that passes different inputs.

INCLUDE Irvine32.inc

DifferentInputs PROTO, v1:DWORD, v2:DWORD, v3:DWORD

.code
main PROC
    INVOKE DifferentInputs, 10, 20, 30
    call WriteDec           ; 1
    call Crlf

    INVOKE DifferentInputs, 10, 10, 30
    call WriteDec           ; 0
    call Crlf

    INVOKE DifferentInputs, 10, 20, 10
    call WriteDec           ; 0
    call Crlf

    INVOKE DifferentInputs, 10, 20, 20
    call WriteDec           ; 0
    call Crlf

    INVOKE DifferentInputs, 5, 4, 3
    call WriteDec           ; 1
    call Crlf

    exit
main ENDP

DifferentInputs PROC, v1:DWORD, v2:DWORD, v3:DWORD
    mov eax, v1
    cmp eax, v2
    je EqualFound

    cmp eax, v3
    je EqualFound

    mov eax, v2
    cmp eax, v3
    je EqualFound

    mov eax, 1              ; 모두 다름
    ret

EqualFound:
    mov eax, 0              ; 같은 값 존재
    ret
DifferentInputs ENDP
END main

7. Greatest Common Divisor
Write a recursive implementation of Euclid’s algorithm for finding the greatest common divisor (GCD) of two integers. Descriptions of this algorithm are available in algebra books and on the Web. Write a test program that calls your GCD procedure five times, using the following pairs of integers: (5,20), (24,18), (11,7), (432,226), (26,13). After each procedure call, display the GCD.

INCLUDE Irvine32.inc

Swap PROTO, p1:PTR DWORD, p2:PTR DWORD

.data
array DWORD 10, 20, 30, 40, 50, 60, 70, 80

.code
main PROC
    mov esi, OFFSET array
    mov ecx, LENGTHOF array / 2

SwapLoop:
    mov edi, esi
    add edi, 4
    INVOKE Swap, esi, edi
    add esi, 8
    loop SwapLoop

    exit
main ENDP

Swap PROC USES eax ebx esi edi, p1:PTR DWORD, p2:PTR DWORD
    mov esi, p1
    mov edi, p2
    mov eax, [esi]
    mov ebx, [edi]
    mov [esi], ebx
    mov [edi], eax
    ret
Swap ENDP
END main

8. Counting Matching Elements
Write a procedure named CountMatches that receives points to two arrays of signed double words, and a third parameter that indicates the length of the two arrays. For each element xi in the first array, if the corresponding yi in the second array is equal, increment a count. At the end, return a count of the number of matching array elements in EAX. Write a test program that calls your procedure and passes pointers to two different pairs of arrays. Use the INVOKE statement to call your procedure and pass stack parameters. Create a PROTO declaration for Count
Matches. Save and restore any registers (other than EAX) changed by your procedure.

INCLUDE Irvine32.inc

CountMatches PROTO, pArr1:PTR SDWORD, pArr2:PTR SDWORD, count:DWORD

.data
arr1 SDWORD 10, 20, 30, 40, 50
arr2 SDWORD 10, 25, 30, 45, 50

.code
main PROC
    INVOKE CountMatches, ADDR arr1, ADDR arr2, LENGTHOF arr1
    call WriteDec           ; 3 (10, 30, 50 일치)
    call Crlf

    exit
main ENDP

CountMatches PROC USES ecx esi edi ebx, pArr1:PTR SDWORD, pArr2:PTR SDWORD, count:DWORD
    mov esi, pArr1
    mov edi, pArr2
    mov ecx, count
    mov ebx, 0              ; 매칭 카운터

L1:

    cmp eax, [edi]
    jne Skip
    inc ebx

Skip:

    add edi, 4
    loop L1

    mov eax, ebx
    ret
CountMatches ENDP
END main


9. Counting Nearly Matching Elements
Write a procedure named CountNearMatches that receives pointers to two arrays of signed doublewords, a parameter that indicates the length of the two arrays, and a parameter that indicates the maximum allowed difference (called diff) between any two matching elements. For each element xi in the first array, if the difference between it and the corresponding yi in the second array is less than or equal to diff, increment a count. At the end, return a count of the number of nearly matching array elements in EAX. Write a test program that calls CountNearMatches and passes pointers to two different pairs of arrays. Use the INVOKE statement to call your procedure and pass stack parameters. Create a PROTO declaration for CountMatches. Save and restore any registers (other than EAX) changed by your procedure.

INCLUDE Irvine32.inc

CountNearMatches PROTO, pArr1:PTR SDWORD, pArr2:PTR SDWORD, count:DWORD, diff:DWORD

.data
a1 SDWORD 10, 20, 30, 40
a2 SDWORD 12, 18, 35, 40

.code
main PROC
    INVOKE CountNearMatches, ADDR a1, ADDR a2, LENGTHOF a1, 3
    call WriteDec           ; 3 (|10-12|=2<=3, |20-18|=2<=3, |40-40|=0<=3)
    call Crlf

    exit
main ENDP

CountNearMatches PROC USES ecx esi edi ebx edx, pArr1:PTR SDWORD, pArr2:PTR SDWORD, count:DWORD, diff:DWORD
    mov esi, pArr1
    mov edi, pArr2
    mov ecx, count
    mov ebx, 0              ; 카운터

L1:
    mov eax, [esi]
    sub eax, [edi]          ; eax = x - y
    jge CheckDiff
    neg eax                 ; 절댓값 구하기 (|x - y|)

CheckDiff:
    cmp eax, diff
    jg Skip
    inc ebx

Skip:
    add esi, 4
    add edi, 4
    loop L1

    mov eax, ebx
    ret
CountNearMatches ENDP
END main

10. Show Procedure Parameters
Write a procedure named ShowParams that displays the address and hexadecimal value of the 32-bit parameters on the runtime stack of the procedure that called it. The parameters are to be displayed in order from the lowest address to the highest. Input to the procedure will be a single integer that indicates the number of parameters to display. For example, suppose the following statement in main calls MySample, passing three arguments: 
	INVOKE MySample, 1234h, 5000h, 6543h
Next, inside MySample, you should be able to call ShowParams, passing the number of param-eters you want to display:
	MySample PROC first:DWORD, second:DWORD, third:DWORD
	paramCount = 3
	call ShowParams, paramCount
ShowParams should display output in the following format:
	Stack parameters:
	--------------------------
	Address 0012FF80 = 00001234
	Address 0012FF84 = 00005000
	Address 0012FF88 = 00006543

INCLUDE Irvine32.inc

ShowParams PROTO, paramCount:DWORD
MySample PROTO, first:DWORD, second:DWORD, third:DWORD

.code
main PROC
    INVOKE MySample, 1234h, 5000h, 6543h
    exit
main ENDP

MySample PROC, first:DWORD, second:DWORD, third:DWORD
    INVOKE ShowParams, 3
    ret
MySample ENDP

ShowParams PROC USES eax ebx ecx edx esi, numParams:DWORD
    mov ebp, [ebp]          ; 호출자(MySample)의 EBP를 가져옴
    mov esi, ebp
    add esi, 8              ; 첫 번째 매개변수 위치 [EBP + 8]
    mov ecx, numParams

    mov edx, OFFSET header
    call WriteString
    call Crlf

L1:
    mov edx, OFFSET addrStr
    call WriteString
    mov eax, esi
    call WriteHex           ; 매개변수 주소 출력

    mov edx, OFFSET valStr
    call WriteString
    mov eax, [esi]
    call WriteHex           ; 매개변수 값 출력
    call Crlf

    add esi, 4              ; 다음 매개변수 주소로 이동
    loop L1

    ret

.data
header  BYTE "Stack parameters:", 0
addrStr BYTE "Address ", 0
valStr  BYTE " = ", 0

ShowParams ENDP
END main
